src/app.ts //

import express, { Application as ExApplication } from 'express';


class Application {
    private readonly _instance: ExApplication;
    get instance(): ExApplication {
        return this._instance;
    };
    constructor() {
        this._instance = express();
        this._instance.use(express.json({ limit: "150mb" }));
        this._instance.use(express.urlencoded({ limit: "150mb", extended: true }));
      
    };

   
}


export default new Application()

  server.ts

import application from './src/app';
import * as http from 'http';
import {Server as S} from 'http'
const server:S = http.createServer(application.instance);
const PORT = 3000
server.listen(PORT, () => {
   
    console.log(`Server started and running on http://localhost:${PORT}} time:${new Date()}`)
});
